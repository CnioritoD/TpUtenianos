Gramatica extraida del PDF del grupo Utenianos

Sigma -> PROGRAMA

PROGRAMA -> INSTRUCCION PROGRAMA
          | INSTRUCCION

INSTRUCCION -> BLOQUE_WHEN
             | BLOQUE_EVERY
             | CONDICIONAL
             | ASIGNACION

BLOQUE_WHEN -> WHEN CONDICION DO LISTA_ACCIONES END

BLOQUE_EVERY -> EVERY TK_TIME DO LISTA_ACCIONES END

CONDICIONAL -> IF CONDICION THEN LISTA_ACCIONES END
             | IF CONDICION THEN LISTA_ACCIONES ELSE LISTA_ACCIONES END

LISTA_ACCIONES -> ACCION LISTA_ACCIONES
                | ACCION

ACCION -> ASIGNACION
        | CONDICIONAL

ASIGNACION -> TK_ID_DISPOSITIVO PUNTO TK_ATRIBUTO IGUAL VALOR

CONDICION -> EXP_LOGICA

EXP_LOGICA -> TERMINO_LOGICO OR EXP_LOGICA
            | TERMINO_LOGICO

TERMINO_LOGICO -> FACTOR_LOGICO AND TERMINO_LOGICO
                | FACTOR_LOGICO

FACTOR_LOGICO -> NOT FACTOR_LOGICO
               | PARENTESIS_ABRE EXP_LOGICA PARENTESIS_CIERRA
               | COMPARACION
               | TK_BOOL
               | TK_ID_SENSOR_BOOL

COMPARACION -> EXP_EVALUADA OP_COMP EXP_EVALUADA

EXP_EVALUADA -> TK_ID_SENSOR_NUM
              | TK_ID_SENSOR_BOOL
              | TK_ID_DISPOSITIVO PUNTO TK_ATRIBUTO
              | VALOR

OP_COMP -> IGUALDAD
         | DIFERENTE
         | MAYOR
         | MENOR
         | MAYOR_IGUAL
         | MENOR_IGUAL

VALOR -> VALOR_NUMERICO
       | TK_BOOL
       | TK_TEXTO
       | TK_HORA
       | TK_FECHA
       | TK_EMAIL
       | TK_COLOR
       | TK_MODO
       | TK_TIME

VALOR_NUMERICO -> TK_NUM_TEMP
                | TK_PERCENT
                | TK_LUX
